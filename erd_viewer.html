            users {
                uuid UserId PK
                varchar Email
                varchar PasswordHash
                boolean IsEmailConfirmed
                varchar OtpCode
                timestamp OtpExpiry
                timestamp CreatedAt
            }

            orders {
                uuid id PK
                uuid customer_id FK
                varchar customer_name
                varchar customer_email
                varchar customer_phone
                varchar customer_id_card
                uuid store_id
                numeric total_amount
                numeric deposit_amount
                varchar deposit_type
                varchar battery_option
                varchar order_status
                timestamp created_at
                timestamp last_modified_at
                timestamp paid_at
                timestamp cancelled_at
                text cancellation_reason
            }

            order_items {
                uuid id PK
                uuid order_id FK
                varchar product_id
                varchar product_name
                char assigned_vin
                numeric unit_price
                integer quantity
                numeric line_total
                timestamp vin_assigned_at
                timestamp created_at
                timestamp last_modified_at
            }

            payments {
                uuid id PK
                uuid order_id FK
                varchar transaction_no
                numeric amount
                varchar payment_method
                varchar status
                jsonb gateway_response
                timestamp created_at
                timestamp paid_at
                timestamp failed_at
                timestamp refunded_at
                varchar refund_reason
            }

            AdminUsers {
                uuid Id PK
                varchar Username
                varchar Email
                varchar PasswordHash
                boolean IsActive
                timestamp CreatedAt
            }

            AdminRefreshTokens {
                uuid Id PK
                uuid AdminUserId FK
                varchar Token
                timestamp ExpiresAt
                timestamp RevokedAt
                timestamp CreatedAt
            }

            Roles {
                uuid Id PK
                varchar Code
                varchar Name
                varchar Description
                timestamp CreatedAt
            }

            Permissions {
                uuid Id PK
                varchar Code
                varchar Name
                varchar Description
                timestamp CreatedAt
            }

            AdminUserRoles {
                uuid AdminUserId PK,FK
                uuid RoleId PK,FK
            }

            RolePermissions {
                uuid RoleId PK,FK
                uuid PermissionId PK,FK
            }

            articles {
                uuid id PK
                varchar title
                varchar slug
                text content
                varchar image_thumbnail
                varchar status
                integer view_count
                uuid author_id FK
                timestamp created_at
                timestamp updated_at
            }

            faqs {
                uuid id PK
                varchar question
                text answer
                varchar category
                integer display_order
                timestamp created_at
                timestamp updated_at
            }

            battery_health_reports {
                varchar vin PK
                double max_temp
                double avg_temp
                integer temp_anomalies_count
                integer total_records
                double avg_discharge_rate
                double soh
                boolean is_risky
                varchar analysis_date
                timestamp processed_at
            }

            outbox_events {
                uuid id PK
                varchar event_type
                text payload
                timestamp created_at
                timestamp processed_at
                text error
            }
